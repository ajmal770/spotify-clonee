from django.shortcuts import render, redirect
from django.contrib.auth.hashers import make_password, check_password
from .models import SignupUser
import re

# ---------------------------
# LOGIN PAGE
# ---------------------------
def loginpage(request):
    if request.method == "POST":
        email = request.POST.get('email')
        password = request.POST.get('password')

        try:
            user = SignupUser.objects.get(email=email)
            if check_password(password, user.password):
                request.session['user_id'] = user.id
                request.session['user_email'] = user.email
                return redirect('home')
            else:
                return render(request, 'loginpage.html', {
                    'error': 'Invalid password'
                })
        except SignupUser.DoesNotExist:
            return render(request, 'loginpage.html', {
                'error': 'User not found'
            })

    return render(request, 'loginpage.html')


# ---------------------------
# STEP 1: EMAIL SIGNUP
# ---------------------------
def sing(request):
    if request.method == "POST":
        email = request.POST.get('email')
        if not email:
            return render(request, 'sing.html', {'error': 'Email is required'})

        request.session['email'] = email
        return redirect('singup2')

    return render(request, 'sing.html')


# ---------------------------
# STEP 2: PASSWORD SIGNUP
# ---------------------------
def sing2(request):
    if request.method == "POST":
        password = request.POST.get('password')

        if not validate_password(password):
            return render(request, 'sing2.html', {
                'error': 'Password must be 10+ chars with letters & numbers/special chars'
            })

        request.session['password'] = password
        return redirect('singup3')

    return render(request, 'sing2.html')


# ---------------------------
# STEP 3: PERSONAL INFO
# ---------------------------
def sing3(request):
    if request.method == "POST":
        name = request.POST.get('name')
        year = request.POST.get('year')
        month = request.POST.get('month')
        day = request.POST.get('day')
        gender = request.POST.get('gender')

        if not (name and year and month and day and gender):
            return render(request, 'sing3.html', {'error': 'All fields required'})

        months = {
            'January': '01', 'February': '02', 'March': '03', 'April': '04',
            'May': '05', 'June': '06', 'July': '07', 'August': '08',
            'September': '09', 'October': '10', 'November': '11', 'December': '12'
        }

        month_number = months.get(month)
        if not month_number:
            return render(request, 'sing3.html', {'error': 'Invalid month'})

        dob = f"{year}-{month_number}-{day.zfill(2)}"

        request.session['name'] = name
        request.session['dob'] = dob
        request.session['gender'] = gender

        return redirect('singup4')

    return render(request, 'sing3.html')


# ---------------------------
# STEP 4: SAVE USER
# ---------------------------
def sing4(request):
    if request.method == "POST":
        email = request.session.get('email')
        password = request.session.get('password')
        name = request.session.get('name')
        dob = request.session.get('dob')
        gender = request.session.get('gender')

        if not (email and password and name and dob and gender):
            return redirect('sing')

        SignupUser.objects.create(
            email=email,
            password=make_password(password),
            name=name,
            dob=dob,
            gender=gender,
            marketing_opt_out=bool(request.POST.get('marketing_opt_out')),
            marketing_share=bool(request.POST.get('marketing_share')),
        )

        request.session.flush()
        return redirect('loginpage')

    return render(request, 'sing4.html')


# ---------------------------
# PASSWORD VALIDATION
# ---------------------------
def validate_password(password):
    if not password:
        return False

    has_letter = re.search(r'[A-Za-z]', password)
    has_number_or_special = re.search(r'[0-9#?!&]', password)
    has_min_length = len(password) >= 10

    return bool(has_letter and has_number_or_special and has_min_length)


# ---------------------------

def home(request):
    if not request.session.get('user_id'):
        return redirect('loginpage')
    return render(request, 'list.html')

from django.db import models

class SignupUser(models.Model):
    email = models.EmailField(unique=True)
    password = models.CharField(max_length=128)
    name = models.CharField(max_length=100)
    dob = models.DateField()
    gender = models.CharField(max_length=20)
    marketing_opt_out = models.BooleanField(default=False)
    marketing_share = models.BooleanField(default=False)

    def __str__(self):
        return self.email
